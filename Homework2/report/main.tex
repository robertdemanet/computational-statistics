\section{Esercizio 1}
Assumiamo di avere le seguenti coordinate nello spazio bidimensionale:
\begin{equation*}
    \text{\textbf{s}} = (s_1, s_2) \in \mathbb{R}^2,
\end{equation*}
e di avere il seguente modello:
\begin{equation*}
    Y(\text{\textbf{s}})|W(\text{\textbf{s}}) \overset{\text{iid}}{\sim} GP(W(\text{\textbf{s}}), \tau^2),
\end{equation*}
\begin{equation*}
     W(\text{\textbf{s}}) \sim GP(m(\text{\textbf{s}}), C(||\text{\textbf{s}} - \text{\textbf{s'}}||; \phi, \sigma^2 )),
\end{equation*}
dove:
\begin{equation*}
    m(\text{\textbf{s}}) = \beta_0 + \beta_1 s_1,
\end{equation*}
e:
\begin{equation*}
    C(||\text{\textbf{s}} - \text{\textbf{s'}}||; \phi, \sigma^2 ) = \sigma^2 \exp{(-\phi ||\text{\textbf{s}} - \text{\textbf{s'}}||)}.
\end{equation*}
Utilizziamo i seguenti parametri:
\begin{equation*}
    \tau^2 = 0.5, \ \ \sigma^2 = 5, \ \ \phi = \frac{3}{10}, \ \ \beta_0 = 2, \ \ \beta_1 = 0.1.
\end{equation*}

\begin{equation*}

\end{equation*}

\subsection{}
Ora vogliamo scegliere un qualsiasi \(n_{new} \in [10,90]\), e selezionare in maniera casuale \(n_{new}\) osservazioni dalle \(n\) osservazioni originali di \(Y(\mathbf{s})|W(\mathbf{s})\). Raccogliamo tali osservazioni nel vettore \(\mathbf{y_o}\) e le restanti nel vettore \(\mathbf{y_u}\), e le coordinate corrispondenti a \(\mathbf{y_o}\) in \(D_o\) e quelle corrispondenti a \(\mathbf{y_u}\) in \(D_u\). \\
Vogliamo definire delle prior appropriate per i parametri e scrivere un algoritmo MCMC per ottenere campioni da:
\begin{equation*}
    f(\mathbf{w}, \beta_0, \beta_1, \tau^2, \sigma^2, \phi|\mathbf{y_o}).
\end{equation*}
Il DAG corrispondente è:
\begin{center}

Definiamo le seguenti prior:
\[ \boldsymbol{\beta} \sim \mathcal{N}_2\left(\mathbf{0}, \mathbf{I}_2\right), \ \sigma^2 \sim IG(1,1), \ \phi \sim IG(1,1),  \]
\[\tau^2 \sim IG(1,1), \ \mathbf{w} \sim \mathcal{N}_{n_{new}}(m(\text{\textbf{s}}), C(||\text{\textbf{s}} - \text{\textbf{s'}}||; \phi, \sigma^2 )),\]
e inizializziamo i dati introdotti in precedenza:
\begin{lstlisting}[style=R]
n_new = runif(1, 10, 90)
n_new = floor(n_new)
indices <- sample(1:100, n_new) 
indices = sort(indices)

y_o = data_sim$Y_s[indices]
D_o = data_sim[indices, c('x', 'y')]

y_u = data_sim$Y_s[-indices]
D_u = data_sim[-indices, c('x', 'y')]
\end{lstlisting}
Stabiliamo il numero di iterazioni dell'algoritmo MCMC e inizializziamo i vettori che andranno a raccogliere le osservazioni a posteriori dei parametri, stabilendo per ciascun vettore dei valori iniziali:
\begin{lstlisting}[style=R]
n_iter = 20000

# Parametri iniziali
beta0_samples = rep(NA, n_iter)
beta0_samples[1] = 0
beta1_samples = rep(NA, n_iter)
beta1_samples[1] = 0
tau2_samples = rep(NA, n_iter)
tau2_samples[1] = 1
sigma2_samples = rep(NA, n_iter)
sigma2_samples[1] = 1
phi_samples = rep(NA, n_iter)
phi_samples[1] = 0.1
W_samples = matrix(NA, nrow = n_iter, ncol = n_new)
W_samples[1, ] = rep(0,n_new)
\end{lstlisting}
Selezioniamo i valori delle coordinate x (\(s_1\)) corrispondenti alle osservazioni selezionate e costruiamo la nuova matrice delle distanze relativa alle osservazioni selezionate:
\begin{lstlisting}[style=R]
D_o_x = data_sim[indices, c('x')]
distance_matrix_o = as.matrix(dist(D_o))
\end{lstlisting}
e costruiamo anche la nuova matrice delle covariate \(\mathbf{X}\):
\begin{lstlisting}[style=R]
ones = rep(1, n_new)
X = cbind(ones, D_o_x)
\end{lstlisting}
Ora siamo pronti a iniziare con le iterazioni dell'algoritmo MCMC:
\begin{lstlisting}[style=R]
for (i in 2:n_iter) 
{
    # ...
}
\end{lstlisting}
Il primo passo dell'algoritmo MCMC consiste nell'aggiornare il valore a posteriori \(\boldsymbol{\beta}|\mathbf{w}\). \\
Sappiamo dalla teoria che:
\begin{align*}
    \boldsymbol{\beta}|\mathbf{w} &\sim \mathcal{N}_2(M_2,V_2), \\
    V_2 &= \left( \mathbf{X}^T C^{-1} \mathbf{X} + \begin{pmatrix}
\frac{1}{100} & 0 \\
0 & \frac{1}{100}
\end{pmatrix} \right) \\
    M_2 &= V_2 \left( C^{-1} \mathbf{X}^T \mathbf{w} + \begin{pmatrix}
0 \\
0
\end{pmatrix} \begin{pmatrix}
\frac{1}{100} & 0 \\
0 & \frac{1}{100}
\end{pmatrix}  \right)
\end{align*}
Calcolo la matrice di covarianza \(C_o(\cdot)\) corrispondente alle osservazioni selezionate e ne calcolo l'inversa:
\begin{lstlisting}[style=R]
C_o = sigma2_samples[i-1] * exp(- phi_samples[i-1] * distance_matrix_o)
inv_C_o = solve(C_o)
\end{lstlisting}
e procedo al campionamento di \(\boldsymbol{\beta}\):
\begin{lstlisting}[style=R]
## Passo (1): aggiornamento beta
# Calcolo Vp
Vp = solve(t(X) %*% inv_C_o %*% X + diag_matrix)
  
# Calcolo Mp
Mp = Vp %*% (t(X) %*% inv_C_o %*% W_samples[i-1,])
  
# Campiono beta
beta = mvrnorm(1, Mp, Vp)
beta0_samples[i] = beta[1]
beta1_samples[i] = beta[2]
\end{lstlisting}
Il secondo passo consiste nell'aggiornare il valore a posteriori \(\sigma^2|\mathbf{w}\). \\
Utilizzando il metodo del kernel, ne troviamo la distribuzione a posteriori:
\begin{align*}
    f(\sigma^2|\mathbf{w}) &\propto f(\mathbf{w}|\sigma^2)f(\sigma^2) \\
    &\propto (\sigma^2)^{-\left( \frac{n_{new}}{2} + a + 1 \right)} \exp{\left\{ - \frac{\left[ \frac{(\mathbf{w} - \mathbf{X} \boldsymbol{\beta})^T C^{-1} (\mathbf{w} - \mathbf{X} \boldsymbol{\beta})}{2} + b \right]}{\sigma^2} \right\}},
\end{align*}
cioè:
\begin{equation*}
    \sigma^2|\mathbf{w} \sim IG\left(\frac{n_{new}}{2} + a, \frac{(\mathbf{w} - \mathbf{X} \boldsymbol{\beta})^T C^{-1} (\mathbf{w} - \mathbf{X} \boldsymbol{\beta})}{2} + b \right),
\end{equation*}
dove nel nostro caso \(a = b = 1\).\\
Procedo quindi al campionamento di \(\sigma^2\):
\begin{lstlisting}[style=R]
## Passo (2): aggiornamento sigma2
a_sigma2 = n_new/2 + 1
prod = t(W_samples[i-1,] - X %*% beta) %*% inv_C_o %*% (W_samples[i-1,] - X %*% beta)
b_sigma2 = prod/2 + 1
  
# Campiono sigma2
sigma2_samples[i] = 1 / rgamma(1, a_sigma2, b_sigma2)
\end{lstlisting}
Il terzo passo consiste nell'aggiornare il valore a posteriori \(\tau^2|\mathbf{y_o}\). \\
Utilizzando il metodo del kernel, ne troviamo la distribuzione a posteriori:
\begin{align*}
    f(\tau^2|\mathbf{y_o}) &\propto f(\mathbf{y_o}|\tau^2)f(\tau^2) \\
    &\propto (\tau^2)^{-\left( \frac{n_{new}}{2} + a + 1 \right)} \exp{\left\{ - \frac{\left[ \frac{(\mathbf{y_o - \mathbf{w}})^T (\mathbf{y_o - \mathbf{w}})}{2} + b \right]}{\tau^2} \right\}},
\end{align*}
cioè:
\begin{equation*}
    \tau^2|\mathbf{y_o} \sim IG\left(\frac{n_{new}}{2} + a, \frac{(\mathbf{y_o} - \mathbf{w})^T (\mathbf{y_o} - \mathbf{w})}{2} + b \right),
\end{equation*}
dove nel nostro caso \(a = b = 1\).\\
Procedo quindi al campionamento di \(\tau^2\):
\begin{lstlisting}[style=R]
## Passo (3): aggiornamento tau2
a_tau2 = n_new/2 + 1
b_tau2 = 0.5 * t(y_o - W_samples[i-1,]) %*% (y_o - W_samples[i-1,]) + 1
  
# Campiono tau2
tau2_samples[i] = 1 / rgamma(1, a_tau2, b_tau2)
\end{lstlisting}
Il quarto passo consiste nell'aggiornare il valore a posteriori \(\phi|\mathbf{w}\). \\
Siccome la forma della suddetta distribuzione è difficile da trovare in forma chiusa, utilizziamo l'algoritmo Metropolis. \\
Si propone un nuovo valore per il parametro \(\phi\), e il valore proposto, denotato come \(\phi_{prop}\), viene generato tramite una distribuzione normale centrata sul valore di \(\phi\) all'i-esima iterazione, con deviazione standard adattiva (inizializzata al valore 0.1). \\
Dopo aver proposto il nuovo valore, si verifica che \(\phi_{prop}\) sia compreso tra due limiti calcolati dalle distanze spaziali tra i punti.
Se il valore proposto non soddisfa questa condizione, viene scartato: 
\begin{lstlisting}[style=R]
## Passo (4): aggiornamento phi (Metropolis)
sd_proposal = 0.01
phi_prop = rnorm(1, phi_samples[i-1], sd_proposal)
if (phi_prop >= 3/max(dist(points)) && phi_prop <= 3/min(dist(points))) 
{
    # ...
}
\end{lstlisting}
Successivamente, si calcola il logaritmo del rapporto di probabilità tra la funzione di verosimiglianza e il log-prior del parametro \(\phi\), dato il valore attuale e il valore proposto. Questo log-ratio è dato da:
\begin{lstlisting}[style=R]
log_ratio = log_fc_phi(phi_prop, W_samples[i-1], beta, sigma2_samples[i], X, C_o) - log_fc_phi(phi_samples[i-1], W_samples[i-1], beta, sigma2_samples[i], X, C_o)
\end{lstlisting}
La funzione \texttt{log\_fc\_phi()} calcola il logaritmo della funzione di verosimiglianza e del log-prior di \texttt{phi}. La verosimiglianza è basata sul determinante della matrice di covarianza \(C\) e sul residuo tra le osservazioni e le predizioni, mentre il log-prior assume una distribuzione Inverse Gamma per \(\phi\). Se il log-ratio è positivo, la proposta viene accettata con maggiore probabilità:
\begin{lstlisting}[style=R]
log_fc_phi = function(phi, w, beta, sigma2, X, C)
{
  # Calcolo del determinante
  det_C = determinant(C, logarithm = TRUE)$modulus  # logaritmo del determinante
  
  det_C = abs(det_C)  # Usato abs nel caso il determinante sia negativo
  
  # Calcolo della log-likelihood
  res = w - X %*% beta
  log_likelihood = -0.5 * (det_C + t(res) %*% solve(C) %*% res)
  
  # Calcolo del log-prior di phi usando la distribuzione Gamma
  log_prior = dgamma(phi, 1, 1, log = TRUE)
  
  return(log_likelihood + log_prior)
}
\end{lstlisting}
La probabilità di accettazione (inizialmente fissata a 0) è data da:
\begin{lstlisting}[style=R]
min(exp(log_ratio), 1)
\end{lstlisting}
Se un numero casuale uniforme tra 0 e 1 è minore della probabilità di accettazione, il valore proposto \(\phi_{prop}\) viene accettato come nuovo valore di \(\phi\):
\begin{lstlisting}[style=R]
if (runif(1, 0, 1) < exp(log_ratio))
{
    phi = phi_prop
}
\end{lstlisting}
Altrimenti, il valore di \(\phi\) rimane invariato. Ogni 50 iterazioni, la deviazione standard della proposta viene adattata per mantenere la frequenza di accettazione delle proposte vicino al valore target 0.234. Questo adattamento viene effettuato nel seguente modo:
\begin{lstlisting}[style=R]
nbatch = 50
A = 100
B = 1000
alpha_target = 0.234

if (i%%nbatch == 0)
{
    alpha_phi = alpha_phi / nbatch
    sd_proposal = exp(log(sd_proposal) + A / (B + i) * (alpha_phi - alpha_target))
    alpha_phi = 0
}
\end{lstlisting}
Infine, il valore aggiornato di \texttt{phi} viene memorizzato come:
\begin{lstlisting}[style=R]
phi_samples[i] = phi
\end{lstlisting}
Il quinto passo consiste nell'aggiornare il valore a posteriori \(\mathbf{w}|\mathbf{y_o}\). \\
Utilizzando il metodo del kernel, ne troviamo la distribuzione a posteriori:
\begin{align*}
    f(\mathbf{w}|\mathbf{y_o}) &\propto f(\mathbf{y_o}|\mathbf{w})f(\mathbf{w}) \\
    &\propto \exp{\left\{ - \frac{\mathbf{w}^T\mathbf{Q}\mathbf{w}}{2} + \mathbf{b}^T \mathbf{w}\right\}},
\end{align*}
dove:
\begin{align*}
    \mathbf{Q} &= \frac{\mathbf{I}}{\tau^2} + C^{-1}, \\
    \mathbf{b} &= \frac{\mathbf{y_o}}{\tau^2} + C^{-1} \mathbf{X} \mathbf{\beta},
\end{align*}
cioè:
\begin{equation*}
    \mathbf{w}|\mathbf{y_o} \sim \mathcal{N}_{n_{new}}(\mathbf{Q}^{-1}\mathbf{b}, \mathbf{Q}^{-1}).
\end{equation*}
Procedo quindi al campionamento di \(\mathbf{w}\):
\begin{lstlisting}[style=R]
# Passo (5): aggiornamento W
  Q = inv_C_o + diag(1 / tau2_samples[i], n_new)
  b = inv_C_o %*% X %*% beta + y_o / tau2_samples[i]
  muW_post = solve(Q) %*% b
  sigmaW_post = solve(Q)
  
  # Campiono W
  W_samples[i,] = mvtnorm::rmvnorm(1, as.vector(muW_post), sigmaW_post)
\end{lstlisting}

\subsection{}
Ora vogliamo ottenere campioni dalla a posteriori di:
\begin{equation*}
    \epsilon(\boldsymbol{s}) = Y(\boldsymbol{s}) - (\beta_0 + \beta_1 s_1), \quad \boldsymbol{s} \in D_o.
\end{equation*}
Come prima cosa, creiamo la matrice che conterrà le osservazioni a posteriori di \(\epsilon(\boldsymbol{s})\) in ciascuna \(\boldsymbol{s}\) osservata:
\begin{lstlisting}[style=R]
epsilon_samples = matrix(NA, nrow = n_iter - burn_in, ncol = n_new)
\end{lstlisting}
Successivamente, iteriamo su tutte le osservazioni a posteriori ottenute dai punti precedenti, ossia quelle dopo il burnin:
\begin{lstlisting}[style=R]
for (i in (burn_in + 1) : n_iter) {
  # ...
}
\end{lstlisting}
All'interno del ciclo costruiamo, utilizzando i samples ottenuti in precedenza, il valore di \(\epsilon(\boldsymbol{s})\):

m_post = beta0_samples[i] + beta1_samples[i] * D_o$x
epsilon_samples[i - burn_in, ] = y_o - m_post
\end{lstlisting}

Ora, vogliamo ottenere campioni dalla a posteriori di:
\begin{equation*}
    f(y(\boldsymbol{s}) | \boldsymbol{y}_o), \quad \boldsymbol{s} \in D_u^* \subseteq D_u.
\end{equation*}
Compongo l'insieme \(D_u^*\) scegliendo 20 coordinate da quelle non osservate:
\begin{lstlisting}[style=R]
n_Du_star = 20
indices_Du_star = sample(1:nrow(D_u), n_Du_star)
D_u_star = D_u[indices_Du_star, ]
true_values = Y_s[indices_Du_star]
\end{lstlisting}
e creo la matrice delle distanze basate su queste coordinate:
\begin{lstlisting}[style=R]
distance_matrix = as.matrix(dist(D_u_star))
\end{lstlisting}
Successivamente, inizializzo la matrice che conterrà le osservazioni a posteriori di  \(f(\cdot)\):
\begin{lstlisting}[style=R]
y_s_samples = matrix(NA, nrow = n_iter - burn_in, ncol = n_Du_star)
\end{lstlisting}
Iteriamo nuovamente sul numero delle osservazioni dopo il burnin:
\begin{lstlisting}[style=R]
for (j in (burn_in + 1) : n_iter) {
  # ...
}
\end{lstlisting}    
All'interno del ciclo calcolo la funzione \(m(\boldsymbol{s})\) utilizzando i dati a posteriori e le ascisse delle coordinate contenute in \(D_u^*\):
\begin{lstlisting}[style=R]
m_s = beta0_samples[j] + D_u_star$x * beta1_samples[j]
\end{lstlisting}  
e procedo a generare campioni da \(Y(\boldsymbol{s})\) esattamente come nel primo punto:
\begin{lstlisting}[style=R]
covariance_matrix = sigma2_samples[j] * exp(- phi_samples[j] * distance_matrix)
W_s = mvrnorm(1, mu = m_s, Sigma = covariance_matrix)
epsilon = rnorm(n_Du_star, 0, sqrt(tau2_samples[j]))
y_s_samples[j - burn_in, ] = W_s + epsilon
\end{lstlisting}  
Procedo a calcolare intervalli di confidenza di livello 95\%:
\begin{lstlisting}[style=R]
conf_intervals = matrix(NA,n_Du_star,2)

for (i in 1:n_Du_star) {
  conf_intervals[i,1] = quantile(y_s_samples[,i], 0.025)
  conf_intervals[i,2] = quantile(y_s_samples[,i], 0.975)

is_inside = sapply(1:n_Du_star, function(i) {

\section{Esercizio 2}
Abbiamo un modello mistura:
\begin{equation*}
    Y_i|z_i \sim P(\lambda_{z_i}),
\end{equation*}
con variabile latente \(\mathbf{z}\) distribuita come:
\begin{equation*}
    \mathbb{P}(z_i = k) = \pi_k.
\end{equation*}
dove abbiamo \(i = 1,...,200\), \(z_i \in \{1,2,3\}\), \(\pi_k = \frac{1}{3}, k = 1,2,3\), e:
\begin{equation*}
    \lambda_1 = 1, \ \lambda_2 = 10, \ \lambda_3 = 25.
\end{equation*}

\subsection{}
Vogliamo ora implementare un metodo MCMC per ottenere delle stime a posteriori di:
\begin{equation*}
    f(\lambda_1,\lambda_2,\lambda_3,\pi_1,\pi_2,\pi_3,z_1,...,z_n|\mathbf{y}).
\end{equation*}
Utilizzeremo le seguenti prior:
\begin{equation*}
    \lambda_k \sim \text{Gamma}(a_{prior},b_{prior}), \ \  \boldsymbol{\pi} \sim \text{Dir}(\boldsymbol{\alpha_{prior}}), \ \ z_i|\boldsymbol{\pi} \sim \text{Discrete}(\boldsymbol{\pi}),
\end{equation*}
per \(k = 1,2,3\), e \(i = 1,...,200\). \\
Come prima cosa, inizializziamo i parametri:
\begin{lstlisting}[style=R]
n_iter = 10000
k = 3
a_gamma_prior = 1
b_gamma_prior = 1
alpha_prior = c(1,1,1) 
\end{lstlisting}
e le matrici che conterranno ciascun aggiornamento dei parametri:
\begin{lstlisting}[style=R]
lambda_samples = matrix(NA, nrow = n_iter, ncol = k)
pi_samples = matrix(NA, nrow=n_iter, ncol=k) 
zi_samples = matrix(NA, nrow=n_iter, ncol=n)
\end{lstlisting}
Successivamente, definiamo i valori iniziali per il metodo MCMC:
\begin{lstlisting}[style=R]
lambda = rep(mean(Y), k)
pi = rep(1/k, k)
z = sample(1:k, n, replace = TRUE)
\end{lstlisting}
e eseguiamo il ciclo per il numero di iterazioni scelto:
\begin{lstlisting}[style=R]
for (i in 1:n_iter) 
{
    # ...
}
\end{lstlisting}
Il primo passo dell'algoritmo MCMC consiste nell'aggiornare il valore a posteriori \(\lambda_k|\mathbf{y},\mathbf{z}\). \\
Utilizzando il metodo del kernel, ne troviamo la distribuzione a posteriori:
\begin{align*}
    f(\lambda_k|\mathbf{y},\mathbf{z}) &\propto f(\mathbf{y}|\lambda_k,\mathbf{z}) f(\lambda_k) \\
    &\propto \lambda_k^{\sum_{i: z_i = k} y_i + a_{prior} - 1} e^{-(n_k + b_{prior}) \lambda_k},
\end{align*}
cioè:
\begin{equation*}
    \lambda_k|\mathbf{y},\mathbf{z} \sim \text{Gamma}\left(a_{prior} + \sum_{i: z_i = k} y_i, b_{prior} + n_k \right),
\end{equation*}
dove nel nostro caso \(a_{prior} = b_{prior} = 1\).\\
Procedo quindi al campionamento di \(\lambda_k\):
\begin{lstlisting}[style=R]
## Passo (1): aggiornamento lambda_k
for (j in 1:k) 
{
  y_j = y[z == j]
  lambda[j] = rgamma(1, a_gamma_prior + sum(Y_j), b_gamma_prior + length(y_j))
}
  
lambda_samples[i,] = lambda
\end{lstlisting}
Il secondo passo dell'algoritmo MCMC consiste nell'aggiornare il valore a posteriori \(\boldsymbol{\pi}|\mathbf{z},\mathbf{y}\). \\
Utilizzando il metodo del kernel, ne troviamo la distribuzioni a posteriori:
\begin{align*}
    f(\boldsymbol{\pi}|\mathbf{z},\mathbf{y}) &\propto f(\mathbf{y}|\boldsymbol{\pi}, \boldsymbol{z}) f(\boldsymbol{\pi}) \\
    &\propto \prod_{k = 1}^K \pi_k^{\alpha_k + n_k - 1},
\end{align*}
cioè:
\begin{equation*}
    \pi_k|\mathbf{z},\mathbf{y} \sim \text{Dir}(\alpha_k + n_k),
\end{equation*}
per \(k = 1,...,K\), con \(K = 3\). \\
Procedo quindi al campionamento di \(\boldsymbol{\pi}\):
\begin{lstlisting}[style=R]
## Passo (2): aggiornamento pi_k
n_k = table(factor(z, levels = 1:k))
pi = rdirichlet(1, alpha_prior + n_k)
pi_samples[i,] = pi
\end{lstlisting}
Il terzo passo dell'algoritmo MCMC consiste nell'aggiornare il valore a posteriori \(z_i|y_i,\boldsymbol{\lambda},\boldsymbol{\pi}\). \\
Utilizzando il metodo del kernel, ne troviamo la distribuzione a posteriori:
\begin{align*}
    f(z_i|y_i,\boldsymbol{\lambda},\boldsymbol{\pi}) &\propto f(y_i|z_i,\boldsymbol{\pi}) f(z_i|\boldsymbol{\pi}) \\
    &\propto \pi_k \lambda_k^{y_i} e^{-\lambda_k},
\end{align*}
cioè:
\begin{equation*}
    z_i|y_i,\boldsymbol{\lambda},\boldsymbol{\pi} \sim \pi_k P(\lambda_k),
\end{equation*}
dove \(k\) è il gruppo a cui appartiene \(z_i\). \\
Nel campionamento di \(z_i\) calcoliamo la probabilità di \(z_i\) di appartenere a ciascun gruppo e solo dopo procediamo al campionamento:
\begin{lstlisting}[style=R]
## Passo (3): aggiornamento z_i
for (j in 1:n) 
{
    log_prob_non_norm = c(0,0,0)
    log_prob_non_norm[1] = log(pi[1]) + dpois(Y[j], lambda[1], log = TRUE)
    log_prob_non_norm[2] = log(pi[2]) + dpois(Y[j], lambda[2], log = TRUE)
    log_prob_non_norm[3] = log(pi[3]) + dpois(Y[j], lambda[3], log = TRUE)
    c = max(log_prob_non_norm)
    prob = exp(log_prob_non_norm - (c + log(sum(exp(log_prob_non_norm - c)))))
    z[j] = sample(1:k, 1, prob = prob)
}
zi_samples[i,] = z
\end{lstlisting}
